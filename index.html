<script>

  // CẤU HÌNH TRANH: Lưới 3 x 4 = 12 ô

  const COLS = 3;

  const ROWS = 4;

  const TOTAL = COLS * ROWS;

  const PIECE_W = 420 / COLS; // 140px

  const PIECE_H = 672 / ROWS; // 168px


  // Bức tranh cây 7 thói quen bạn đã tải lên (ảnh được nén dạng Base64 hoặc link ảnh trực tiếp)

  const IMG_SRC = 'https://images2.imgbox.com/7b/03/h7GqI7hW_o.jpg'; // Ảnh cây 7 thói quen chuẩn tỉ lệ


  let currentOrder = [];

  let selectedIdx = null;

  let swaps = 0;

  let timer = null;

  let secondsElapsed = 0;

  let currentTeam = "";

  let currentClass = "";


  function startGame() {

    const teamInput = document.getElementById('teamName').value.trim();

    const classInput = document.getElementById('className').value.trim();

    if (!teamInput) {

      alert("Vui lòng nhập tên nhóm hoặc tên của bạn!");

      return;

    }

    currentTeam = teamInput;

    currentClass = classInput || "---";


    document.getElementById('startScreen').style.display = 'none';

    document.getElementById('gameScreen').style.display = 'flex';


    initBoard();

    startTimer();

  }


  function initBoard() {

    swaps = 0;

    document.getElementById('swapCount').innerText = swaps;

    

    // Tạo mảng thứ tự và xáo trộn ngẫu nhiên

    currentOrder = Array.from({length: TOTAL}, (_, i) => i);

    do {

      currentOrder.sort(() => Math.random() - 0.5);

    } while (countCorrect() > 2); // đảm bảo sau khi xáo trộn không bị đúng quá nhiều


    renderGrid();

    updateStatus();

  }


  function renderGrid() {

    const grid = document.getElementById('puzzle-grid');

    grid.innerHTML = '';


    currentOrder.forEach((posIdx, slotIdx) => {

      const piece = document.createElement('div');

      piece.className = 'piece';

      if (selectedIdx === slotIdx) piece.classList.add('selected');


      // Tính toạ độ ảnh của mảnh ban đầu

      const originCol = posIdx % COLS;

      const originRow = Math.floor(posIdx / COLS);

      piece.style.backgroundImage = `url('${IMG_SRC}')`;

      piece.style.backgroundPosition = `-${originCol * PIECE_W}px -${originRow * PIECE_H}px`;


      piece.onclick = () => onPieceClick(slotIdx);

      grid.appendChild(piece);

    });

  }


  function onPieceClick(slotIdx) {

    if (selectedIdx === null) {

      selectedIdx = slotIdx;

      renderGrid();

    } else if (selectedIdx === slotIdx) {

      selectedIdx = null;

      renderGrid();

    } else {

      // Đổi chỗ hai mảnh

      const temp = currentOrder[selectedIdx];

      currentOrder[selectedIdx] = currentOrder[slotIdx];

      currentOrder[slotIdx] = temp;


      swaps++;

      document.getElementById('swapCount').innerText = swaps;

      selectedIdx = null;

      renderGrid();

      updateStatus();


      if (countCorrect() === TOTAL) {

        finishGame();

      }

    }

  }


  function countCorrect() {

    return currentOrder.reduce((acc, curr, idx) => acc + (curr === idx ? 1 : 0), 0);

  }


  function updateStatus() {

    document.getElementById('correctCount').innerText = `${countCorrect()}/${TOTAL}`;

  }


  function startTimer() {

    secondsElapsed = 0;

    clearInterval(timer);

    timer = setInterval(() => {

      secondsElapsed++;

      const m = String(Math.floor(secondsElapsed / 60)).padStart(2, '0');

      const s = String(secondsElapsed % 60).padStart(2, '0');

      document.getElementById('timerVal').innerText = `${m}:${s}`;

    }, 1000);

  }


  function finishGame() {

    clearInterval(timer);

    setTimeout(() => {

      saveResult();

      showLeaderboard();

    }, 400);

  }


  function saveResult() {

    const results = JSON.parse(localStorage.getItem('puzzle_leaderboard') || '[]');

    results.push({

      name: currentTeam,

      class: currentClass,

      time: secondsElapsed,

      swaps: swaps,

      date: new Date().toLocaleTimeString()

    });

    // Sắp xếp: Ưu tiên thời gian ít hơn, nếu bằng nhau xét ít lần đổi hơn

    results.sort((a, b) => a.time - b.time || a.swaps - b.swaps);

    localStorage.setItem('puzzle_leaderboard', JSON.stringify(results));

  }


  function showLeaderboard() {

    document.getElementById('gameScreen').style.display = 'none';

    document.getElementById('resultScreen').style.display = 'block';


    const results = JSON.parse(localStorage.getItem('puzzle_leaderboard') || '[]');

    const tbody = document.getElementById('rankBody');

    tbody.innerHTML = '';


    results.forEach((r, idx) => {

      const m = String(Math.floor(r.time / 60)).padStart(2, '0');

      const s = String(r.time % 60).padStart(2, '0');

      const tr = document.createElement('tr');

      tr.innerHTML = `

        <td><strong>#${idx + 1}</strong></td>

        <td>${r.name}</td>

        <td>${r.class}</td>

        <td>${m}:${s}</td>

        <td>${r.swaps} lần</td>

      `;

      tbody.appendChild(tr);

    });

  }


  function resetGame() {

    document.getElementById('resultScreen').style.display = 'none';

    document.getElementById('startScreen').style.display = 'block';

  }

</script>
